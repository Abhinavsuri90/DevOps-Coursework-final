# 2. First attempt in ap-south-1 (Mumbai): plan works, apply is DENIED
$ grep aws_region terraform.tfvars

  # aws_s3_bucket_public_access_block.demo will be created
  # aws_s3_bucket_server_side_encryption_configuration.demo will be created
  # aws_s3_bucket_versioning.demo will be created

$ terraform apply -input=false -no-color demo.tfplan 2>&1 | grep -E 'Error:|explicit deny' | cut -c1-140 | mask
Error: creating S3 Bucket (abhinav-24bcs10282-tf-s3-demo): operation error S3: CreateBucket, https response error StatusCode: 403, RequestID
# Root cause: AWS Free-plan accounts sit in an AWS-managed organization whose service control policy only allows
# the account's project region. Proof: a throwaway bucket per region (created, then deleted at once)
$ B=abhinav-region-check-$(openssl rand -hex 3); for r in ap-southeast-2 ap-south-1 us-east-1; do if aws s3api create-bucket --bucket $B-$r --region $r --create-bucket-configuration LocationConstraint=$r >/dev/null 2>&1 || { [ $r = us-east-1 ] && aws s3api create-bucket --bucket $B-$r --region $r >/dev/null 2>&1; }; then echo "$r: allowed"; aws s3api delete-bucket --bucket $B-$r --region $r; else echo "$r: denied by the service control policy"; fi; done
ap-southeast-2: allowed
ap-south-1: denied by the service control policy
us-east-1: denied by the service control policy
